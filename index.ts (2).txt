export interface Env {
  DB: D1Database;
  ADMIN_TOKEN: string;
}
function generateId(): string {
  const y = new Date().getFullYear();
  const r = Math.random().toString(36).substring(2,7).toUpperCase();
  return `PS-VOL-${y}-${r}`;
}
function cors() {
  return {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "GET, POST, PUT, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type, Authorization, X-Admin-Token",
  };
}
export default {
  async fetch(req: Request, env: Env): Promise<Response>{
  const url = new URL(req.url);
  const headers = cors();
  if(req.method === "OPTIONS") return new Response(null,{headers});
  if(url.pathname === "/api/volunteer/register" && req.method === "POST"){
  try{
    const d: any = await req.json();
    if(!d.name || !d.father_name || !d.mobile || !d.full_address){
      return Response.json({success:false,message:"নাম, পিতার নাম, মোবাইল, ঠিকানা বাধ্যতামূলক"}, {status:400, headers});
    }
    const mobile = d.mobile.toString().trim().replace(/\s+/g,'');
    if(!/^[6-9]\d{9}$/.test(mobile)){
      return Response.json({success:false,message:"সঠিক ১০ সংখ্যার মোবাইল নম্বর দিন"}, {status:400, headers});
    }
    const exists = await env.DB.prepare("SELECT id FROM volunteers WHERE mobile = ?").bind(mobile).first();
    if(exists){
      return Response.json({success:false,message:"এই মোবাইল নম্বর দিয়ে আগেই আবেদন করা হয়েছে!", existing_id: exists.id}, {headers});
    }
    const id = generateId();
    await env.DB.prepare("INSERT INTO volunteers (id,name,father_name,dob,gender,mobile,whatsapp,full_address,district,assembly) VALUES (?,?,?,?,?,?,?,?,?,?)").bind(id, d.name, d.father_name, d.dob||'', d.gender||'', mobile, d.whatsapp||'', d.full_address, d.district||'', d.assembly||'').run();
    return Response.json({success:true,id, message:"আবেদন সফল!"}, {headers});
  }catch(e:any){
    if(e.message?.includes("UNIQUE")) return Response.json({success:false,message:"এই মোবাইল নম্বর আগেই ব্যবহার হয়েছে"}, {headers});
    return Response.json({success:false,message:"Server Error: "+e.message}, {status:500, headers});
  }
  }
  if(url.pathname.startsWith("/api/volunteer/status/")){
    const q = decodeURIComponent(url.pathname.split("/").pop()||"").trim();
    const row = await env.DB.prepare("SELECT id,mobile,status,created_at,district,assembly FROM volunteers WHERE id = ? OR mobile = ?").bind(q,q).first();
    if(!row) return Response.json({success:false,message:"ID বা মোবাইল নম্বর পাওয়া যায়নি"}, {status:404, headers});
    return Response.json({success:true,data:row},{headers});
  }
  if(url.pathname.startsWith("/api/volunteer/data/")){
    const q = decodeURIComponent(url.pathname.split("/").pop()||"").trim();
    const row = await env.DB.prepare("SELECT * FROM volunteers WHERE id = ? OR mobile = ?").bind(q,q).first();
    if(!row) return Response.json({success:false}, {status:404, headers});
    return Response.json({success:true,data:row},{headers});
  }
  if(url.pathname === "/api/admin/volunteers"){
    const token = req.headers.get("X-Admin-Token") || url.searchParams.get("token");
    if(token !== env.ADMIN_TOKEN) return Response.json({success:false,message:"Unauthorized"}, {status:401, headers});
    const { results } = await env.DB.prepare("SELECT * FROM volunteers ORDER BY created_at DESC").all();
    return Response.json({success:true,data:results},{headers});
  }
  if(url.pathname.startsWith("/api/admin/volunteer/") && req.method === "PUT"){
    const token = req.headers.get("X-Admin-Token") || url.searchParams.get("token");
    if(token !== env.ADMIN_TOKEN) return Response.json({success:false,message:"Unauthorized"}, {status:401, headers});
    const id = url.pathname.split("/").pop()||"";
    const b: any = await req.json();
    await env.DB.prepare("UPDATE volunteers SET status = ? WHERE id = ?").bind(b.status, id).run();
    return Response.json({success:true,message:"Status updated"},{headers});
  }
  return new Response("Prasenjit Volunteer API Running", {headers});
  }
}
