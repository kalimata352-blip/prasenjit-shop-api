
-- D1 Database Schema for Prasenjit Volunteer App
DROP TABLE IF EXISTS volunteers;
CREATE TABLE volunteers (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  father_name TEXT NOT NULL,
  dob TEXT,
  gender TEXT,
  mobile TEXT NOT NULL UNIQUE,
  whatsapp TEXT,
  full_address TEXT NOT NULL,
  district TEXT DEFAULT 'বীরভূম',
  assembly TEXT,
  qualification TEXT,
  experience TEXT,
  status TEXT DEFAULT 'pending',
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_mobile ON volunteers(mobile);
CREATE INDEX idx_status ON volunteers(status);
